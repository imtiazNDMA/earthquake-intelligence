# === STEP 1: Import Libraries ===
from google.colab import files
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# === Southern Pakistan (Chaman Fault) Earthquake Parameters ===
MainMag = 7.7  # 1935 Quetta Earthquake
target_mags = [4, 5, 6, 7]  # Same as Northern Pakistan for comparison
days_list = [0, 1, 3, 5, 7, 14, 30]

# Adjusted model constants for Chaman fault system
b = 1.0     # Higher b-value for strike-slip
Mmin = 4.0  # To match Northern Pakistan logic
k = 60.0    # Higher productivity
c = 0.3     # Omori time offset
p = 1.15    # Decay constant

# === Helper Functions ===
def gutenberg_richter_scaling(Mtarget, b=b, Mmin=Mmin):
    if Mtarget < Mmin:
        return 1.0
    else:
        return 10 ** (-b * (Mtarget - Mmin))

def omori_utsu_rate(t, k=k, c=c, p=p):
    return round(k / ((t + c) ** p), 4)

def aftershock_prob_per_day(t, Mtarget):
    lam_total = omori_utsu_rate(t)
    mag_scaling = gutenberg_richter_scaling(Mtarget)
    lam_target = lam_total * mag_scaling
    prob = 1 - np.exp(-lam_target)
    return round(prob * 100, 2)

# === STEP 2: Compute Probabilities ===
results = []
for t in days_list:
    for Mtarget in target_mags:
        prob = aftershock_prob_per_day(t, Mtarget)
        results.append({
            'MainMag': MainMag,
            'DaysSince': t,
            'Mtarget': Mtarget,
            'AftershockProb (%)': prob,
            'OmoriRate': omori_utsu_rate(t)
        })

result_df = pd.DataFrame(results)

# === STEP 3: Plot Graph ===
plt.figure(figsize=(10, 6))
for Mtarget in target_mags:
    subset = result_df[result_df['Mtarget'] == Mtarget]
    plt.plot(subset['DaysSince'], subset['AftershockProb (%)'], 
             marker='o', label=f'M ≥ {Mtarget}')

plt.title('Aftershock Probability (Southern Pakistan - Chaman Fault)')
plt.xlabel('Days Since Mainshock')
plt.ylabel('Aftershock Probability (%)')
plt.grid(True)
plt.legend()
plt.xticks(days_list)
plt.tight_layout()
plt.show()

# === STEP 4: Save to CSV and Download ===
output_filename = "chaman_fault_aftershock_probs_final.csv"
result_df.to_csv(output_filename, index=False)
files.download(output_filename)
